% --- FIX bug de "opciones/colores impresos como texto" (preflight §3/§4) -----
%  beamer carga xcolor SIN [table] → colortbl ausente → \rowcolor{eco!12} sale
%  literal ("eco!12Variable"). Y beamer NO trae enumitem → itemize[opciones]
%  imprime el "[leftmargin=...]" como texto. Se cargan aquí (ambos ya viven en

\usepackage{lmodern}              % math Latin Modern ESCALABLE (design system dice
                                  %   "math en Latin Modern"): sin lmodern, el math
                                  %   CM solo existe en tamaños fijos → avisos
                                  %   "Font shape ... not available in size <NN>" al
                                  %   usar math a tamaños Display. Con lmodern: 0 avisos.
\usepackage{colortbl}             % habilita \rowcolor/\columncolor con tintes !NN
\usepackage{enumitem}             % habilita itemize/enumerate[clave=valor]
\setlist{leftmargin=1.4em, itemsep=3pt, topsep=2pt}  % listas más aireadas por defecto

% \DisplaySB (Inter Display SemiBold) no trae miembro BOLD y fontspec no lo
% autodetecta para un nombre de peso específico → \textbf dentro de \idea/\chipf
% avisa "TU/InterDisplaySemiBold/b/n undefined". Le damos un BoldFont explícito
% (la misma SemiBold: el énfasis ya es grande, sin cambio visual) → 0 avisos.
\renewfontfamily\DisplaySB{Inter Display SemiBold}[Ligatures=TeX,
  BoldFont={Inter Display SemiBold}]

\setsubject{eco}                  % índigo RGB(70,55,130)
\definecolor{ecooro}{RGB}{210,166,60}     % dorado capidólar (solo ornamento)

% --- Chips para notación de dinero/valor (no usan \MakeUppercase) ------
\newcommand{\chip}[1]{\tikz[baseline=(c.base)]{\node[fill=subjectcolor!15,
  rounded corners=2.5pt, inner xsep=5pt, inner ysep=2pt,
  font=\DisplaySB](c){#1};}}
\newcommand{\chipf}[1]{\tikz[baseline=(c.base)]{\node[fill=subjectcolor,
  text=white, rounded corners=2.5pt, inner xsep=5pt, inner ysep=2pt,
  font=\DisplaySB](c){#1};}}

% Titulares grandes NO deben partir palabras (queda feo en video/proyección).
% OJO (preflight): dentro de nodos TikZ con `text width`, subir \hyphenpenalty
% NO evita la hipernación. Lo que SÍ funciona es anular el guion del font ACTIVO
% con \hyphenchar\font=-1 DESPUÉS del \selectfont del tamaño usado → \nh.
\newcommand{\nohyf}{\hyphenpenalty=10000\relax\exhyphenpenalty=10000\relax}
\newcommand{\nh}{\hyphenchar\font=-1\relax}

% --- FOOTLINE sin institución (solo profe · tema · número) -------------
\setbeamertemplate{footline}{%
  \hbox{}\vspace{2pt}\hspace{0.6em}%
  {\footnotesize\color{ink!45}\bmrprofe}%
  \hfill{\footnotesize\color{ink!45}\bmrtema}\hspace{0.6em}%
  \tikz[baseline=-3pt]{\node[fill=subjectcolor, text=white, inner xsep=6pt,
     inner ysep=2.5pt, font=\DisplaySB\scriptsize]{\insertframenumber};}%
  \hspace{0.6em}\vspace{4pt}}
